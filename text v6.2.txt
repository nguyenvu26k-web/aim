local originalGraphics = { Lighting = {}, Parts = {}, Terrain = {} }
local isGraphicsOptimized = false

local function applyLowGraphics()
    if isGraphicsOptimized then return end
    isGraphicsOptimized = true
    
    originalGraphics.Lighting.GlobalShadows = Lighting.GlobalShadows
    Lighting.GlobalShadows = false
    for _, effect in ipairs(Lighting:GetChildren()) do
        if effect:IsA("PostEffect") or effect:IsA("BlurEffect") or effect:IsA("SunRaysEffect") then
            originalGraphics.Lighting[effect] = effect.Enabled
            effect.Enabled = false
        end
    end
    
    for _, part in ipairs(Workspace:GetDescendants()) do
        if part:IsA("BasePart") then
            if not originalGraphics.Parts[part] then
                originalGraphics.Parts[part] = { Material = part.Material, CastShadow = part.CastShadow }
            end
            part.Material = Enum.Material.SmoothPlastic
            part.CastShadow = false
        end
    end
end

local function restoreGraphics()
    if not isGraphicsOptimized then return end
    isGraphicsOptimized = false
    
    Lighting.GlobalShadows = originalGraphics.Lighting.GlobalShadows
    for effect, state in pairs(originalGraphics.Lighting) do
        if effect and effect.Parent then effect.Enabled = state end
    end
    for part, data in pairs(originalGraphics.Parts) do
        if part and part.Parent then
            part.Material = data.Material
            part.CastShadow = data.CastShadow
        end
    end
    originalGraphics = { Lighting = {}, Parts = {}, Terrain = {} }
end

-- ==================== NO RECOIL SYSTEM ====================
local lastCameraLookVector = Camera.CFrame.LookVector
addConn(RunService.RenderStepped:Connect(function()
    if Config.NoRecoil and LocalPlayer.Character then
        local tool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
        if tool then
            for _, v in ipairs(tool:GetDescendants()) do
                if v:IsA("NumberValue") or v:IsA("IntValue") then
                    if string.find(string.lower(v.Name), "recoil") or string.find(string.lower(v.Name), "kick") then
                        v.Value = v.Value * (1 - (Config.RecoilReduction / 100))
                    end
                end
            end
            local currentLook = Camera.CFrame.LookVector
            local pitchDiff = currentLook.Y - lastCameraLookVector.Y
            if pitchDiff > 0.005 then 
                local reductionFactor = Config.RecoilReduction / 100
                local newCFrame = Camera.CFrame * CFrame.Angles(-(pitchDiff * reductionFactor), 0, 0)
                Camera.CFrame = Camera.CFrame:Lerp(newCFrame, 0.5)
            end
        end
    end
    lastCameraLookVector = Camera.CFrame.LookVector
end))

-- ==================== LOGIC ĐỊCH/TA & ESP (PLAYERS) ====================
local function IsEnemy(targetPlayer)
    if targetPlayer == LocalPlayer then return false end
    if not Config.TeamCheck then return true end 
    if LocalPlayer.Team == nil or targetPlayer.Team == nil then return true end
    return LocalPlayer.Team ~= targetPlayer.Team
end

local function getPlayerNameText(player)
    if Config.NameMode == "DisplayName" then return player.DisplayName end
    if Config.NameMode == "Username" then return "@" .. player.Name end
    return player.DisplayName .. " (@" .. player.Name .. ")"
end

local function refreshESP()
    for player, highlight in pairs(activeHighlights) do
        local tag = activeNameTags[player]
        if player and player.Character and player.Character:FindFirstChild("HumanoidRootPart") and highlight.Parent then
            local isEnm = IsEnemy(player)
            local isVisible = isEnm or Config.ShowTeammates
            local color = isEnm and Config.EnemyColor or Config.TeamColor
            
            if Config.ESP and isVisible then highlight.Enabled = true; highlight.FillColor = color else highlight.Enabled = false end
            if tag then
                if Config.NameESP and isVisible then
                    tag.Enabled = true; tag.MaxDistance = Config.NameMaxDistance; tag.TextLabel.TextColor3 = color
                    if not Config.ShowDistance then tag.TextLabel.Text = getPlayerNameText(player) end
                else tag.Enabled = false end
            end
        else
            if highlight then highlight:Destroy() end; if tag then tag:Destroy() end
            activeHighlights[player] = nil; activeNameTags[player] = nil
        end
    end
end

local function applyESP(player, character)
    if not character then return end
    local hrp = character:WaitForChild("HumanoidRootPart", 3)
    local head = character:WaitForChild("Head", 3) or hrp
    if not hrp then return end

    if activeHighlights[player] then activeHighlights[player]:Destroy() end
    if activeNameTags[player] then activeNameTags[player]:Destroy() end
    
    local highlight = Instance.new("Highlight")
    highlight.Name = "TestingESP"; highlight.Adornee = character; highlight.OutlineColor = Color3.new(1, 1, 1); highlight.OutlineTransparency = 0.2
    pcall(function() highlight.Parent = CoreGui end)
    if not highlight.Parent then highlight.Parent = LocalPlayer:WaitForChild("PlayerGui") end
    activeHighlights[player] = highlight
    
    local bbGui = Instance.new("BillboardGui")
    bbGui.Name = "TestingNameESP"; bbGui.Adornee = head; bbGui.Size = UDim2.new(0, 200, 0, 50); bbGui.StudsOffset = Vector3.new(0, 1.5, 0); bbGui.AlwaysOnTop = true; bbGui.MaxDistance = Config.NameMaxDistance
    local textLabel = Instance.new("TextLabel")
    textLabel.Name = "TextLabel"; textLabel.Size = UDim2.new(1, 0, 1, 0); textLabel.BackgroundTransparency = 1; textLabel.Font = Enum.Font.GothamBold; textLabel.TextSize = 13; textLabel.TextColor3 = Color3.new(1, 1, 1); textLabel.TextStrokeTransparency = 0.2; textLabel.Text = getPlayerNameText(player); textLabel.Parent = bbGui
    pcall(function() bbGui.Parent = CoreGui end)
    if not bbGui.Parent then bbGui.Parent = LocalPlayer:WaitForChild("PlayerGui") end
    activeNameTags[player] = bbGui
    refreshESP()
end

    if player == LocalPlayer then return end
    addConn(player:GetPropertyChangedSignal("Team"):Connect(refreshESP))
    addConn(player.CharacterAdded:Connect(function(char) applyESP(player, char) end))
    if player.Character then applyESP(player, player.Character) end
end

-- ==================== LOGIC ESP (NPCs) ====================
local function refreshNpcESP()
    for npc, highlight in pairs(activeNpcHighlights) do
        local tag = activeNpcTags[npc]
        if npc and npc.Parent and npc:FindFirstChild("HumanoidRootPart") and npc:FindFirstChild("Humanoid") and npc.Humanoid.Health > 0 and highlight.Parent then
            if Config.NPC_ESP then
                highlight.Enabled = true; highlight.FillColor = Config.NPCColor
                if tag then
                    if Config.NameESP then
                        tag.Enabled = true; tag.MaxDistance = Config.NameMaxDistance; tag.TextLabel.TextColor3 = Config.NPCColor
                        if not Config.ShowDistance then tag.TextLabel.Text = "[NPC] " .. npc.Name end
                    else tag.Enabled = false end
                end
            else
                highlight.Enabled = false; if tag then tag.Enabled = false end
            end
        else
            if highlight then highlight:Destroy() end; if tag then tag:Destroy() end
            activeNpcHighlights[npc] = nil; activeNpcTags[npc] = nil
        end
    end
end

-- ==================== THUẬT TOÁN NGẮM TỐI ƯU (RAYCAST & NPC) ====================
local AimRayParams = RaycastParams.new()
AimRayParams.FilterType = Enum.RaycastFilterType.Exclude
AimRayParams.IgnoreWater = true 

local function IsTargetVisible(targetPart, targetChar)
    if not LocalPlayer.Character or not Camera then return false end
    AimRayParams.FilterDescendantsInstances = {LocalPlayer.Character, Camera}
    
    local origin = Camera.CFrame.Position
    local direction = targetPart.Position - origin
    local result = workspace:Raycast(origin, direction, AimRayParams)
    
    if result then
        if result.Instance:IsDescendantOf(targetChar) then return true end
        if result.Instance.Transparency >= 1 and not result.Instance.CanCollide then return true end
        return false 
    end
    return true 
end

local function GetBestVisibleTarget()
    local nearestDist = Config.FOVRadius
    local bestTarget = nil
    local screenCenter = Camera.ViewportSize / 2 
    local debugReason = "Idle: Không tìm thấy mục tiêu"

    -- 1. QUÉT NGƯỜI CHƠI
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        local char = player.Character
        if not char then continue end
        local hum, hrp, targetPart = char:FindFirstChild("Humanoid"), char:FindFirstChild("HumanoidRootPart"), char:FindFirstChild(Config.TargetPart)
        
        if not (hum and hrp and targetPart) then continue end
        if hum.Health <= 0 then continue end
        if not IsEnemy(player) then debugReason = "Bỏ qua: Đồng đội"; continue end
        
        local dist3D = (targetPart.Position - Camera.CFrame.Position).Magnitude
        if dist3D > Config.MaxDistance then continue end
        local vector, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
        if not onScreen then continue end
        local dist2D = (Vector2.new(vector.X, vector.Y) - screenCenter).Magnitude
        if dist2D > nearestDist then continue end
        
        if Config.VisibilityCheck and not IsTargetVisible(targetPart, char) then
            debugReason = "Đã bỏ khóa: Kẻ địch nấp sau tường"; continue
        end
        
        nearestDist = dist2D; bestTarget = targetPart; debugReason = "Đang khóa: " .. player.Name
    end
    
    -- 2. QUÉT NPC (Nếu Bật)
    if Config.AimNPC then
        for npc, _ in pairs(activeNpcHighlights) do
            if not npc or not npc.Parent then continue end
            local hum, hrp, targetPart = npc:FindFirstChild("Humanoid"), npc:FindFirstChild("HumanoidRootPart"), npc:FindFirstChild(Config.TargetPart)
            
            if not (hum and hrp and targetPart) then continue end
            if hum.Health <= 0 then continue end
            
            local dist3D = (targetPart.Position - Camera.CFrame.Position).Magnitude
            if dist3D > Config.MaxDistance then continue end
            local vector, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
            if not onScreen then continue end
            local dist2D = (Vector2.new(vector.X, vector.Y) - screenCenter).Magnitude
            if dist2D > nearestDist then continue end
            
            if Config.VisibilityCheck and not IsTargetVisible(targetPart, npc) then
                debugReason = "Đã bỏ khóa: NPC bị che khuất"; continue
            end
            
            nearestDist = dist2D; bestTarget = targetPart; debugReason = "Đang khóa Boss/NPC: " .. npc.Name
        end
    end
    
    Config.DebugText = debugReason
    return bestTarget
end

    Config.CurrentFPS = math.floor(1 / deltaTime)

    FovCircle.Size = UDim2.fromOffset(Config.FOVRadius * 2, Config.FOVRadius * 2); FovCircle.Visible = Config.AimAssist
    if Config.AimAssist then
        currentTarget = GetBestVisibleTarget()
        if currentTarget then
            local aimPosition = currentTarget.Position
            if Config.PredictMovement then
                local velocity, distance = currentTarget.AssemblyLinearVelocity, (aimPosition - Camera.CFrame.Position).Magnitude
                if Config.BulletSpeed > 0 then
                    local travelTime = distance / Config.BulletSpeed
                    aimPosition = aimPosition + (velocity * travelTime)
                    if Config.DropCompensation then aimPosition = aimPosition + Vector3.new(0, 0.5 * workspace.Gravity * math.pow(travelTime, 2), 0) end
                else aimPosition = aimPosition + (velocity * 0.04) end
            end
            Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, aimPosition), math.clamp(Config.Smoothness * (deltaTime * 60), 0.1, 1))
        end
    end
    
    if Config.NameESP and Config.ShowDistance then
        local myPos = Camera.CFrame.Position
        for player, tag in pairs(activeNameTags) do
            if tag.Enabled and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local dist = math.floor((player.Character.HumanoidRootPart.Position - myPos).Magnitude)
                tag.TextLabel.Text = getPlayerNameText(player) .. " [" .. dist .. "]"
            end
        end
        if Config.NPC_ESP then
            for npc, tag in pairs(activeNpcTags) do
                if tag.Enabled and npc:FindFirstChild("HumanoidRootPart") then
                    local dist = math.floor((npc.HumanoidRootPart.Position - myPos).Magnitude)
                    tag.TextLabel.Text = "[NPC] " .. npc.Name .. " [" .. dist .. "]"
                end
            end
        end
    end
end))

local function makeDraggable(guiObj)
    local dragging, dragInput, dragStart, frameStart
    addConn(guiObj.InputBegan:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = true; dragStart = input.Position; frameStart = guiObj.Parent == FovGui and guiObj.Position or MainFrame.Position; addConn(input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)) end end))
    addConn(guiObj.InputChanged:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end end))
    addConn(UserInputService.InputChanged:Connect(function(input) if input == dragInput and dragging then local delta = input.Position - dragStart; local target = guiObj.Parent == FovGui and guiObj or MainFrame; target.Position = UDim2.new(frameStart.X.Scale, frameStart.X.Offset + delta.X, frameStart.Y.Scale, frameStart.Y.Offset + delta.Y) end end))

    addConn(btnMinus.MouseButton1Click:Connect(function() Config[configKey] = math.max(min, Config[configKey] - step); label.Text = text .. ": " .. Config[configKey]; if callback then callback() end end))
    addConn(btnPlus.MouseButton1Click:Connect(function() Config[configKey] = math.min(max, Config[configKey] + step); label.Text = text .. ": " .. Config[configKey]; if callback then callback() end end))
end
